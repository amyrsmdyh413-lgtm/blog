<script>
  const wuduSteps = [
    { id: 'w1', text: 'نیت کردن' },
    { id: 'w2', text: 'شستن صورت' },
    { id: 'w3', text: 'شستن دست راست' },
    { id: 'w4', text: 'شستن دست چپ' },
    { id: 'w5', text: 'مسح سر' },
    { id: 'w6', text: 'مسح پای راست' },
    { id: 'w7', text: 'مسح پای چپ' }
  ];

  const salahSteps = [
    { id: 's1', text: 'تکبیره الاحرام (الله اکبر)' },
    { id: 's2', text: 'خواندن حمد و سوره (قیام)' },
    { id: 's3', text: 'رکوع' },
    { id: 's4', text: 'سجده اول و دوم' },
    { id: 's5', text: 'تشهد و سلام' }
  ];

  const allSteps = [
    ...wuduSteps.map(s => ({ ...s, phase: 'آموزش وضو' })),
    ...salahSteps.map(s => ({ ...s, phase: 'آموزش نماز' }))
  ];

  let currentStepIndex = 0;

  function shuffleArray(array) {
    let currentIndex = array.length, randomIndex;
    while (currentIndex !== 0) {
      randomIndex = Math.floor(Math.random() * currentIndex);
      currentIndex--;
      [array[currentIndex], array[randomIndex]] = [array[randomIndex], array[currentIndex]];
    }
    return array;
  }

  function startGame() {
    document.getElementById('game-intro').classList.add('hidden');
    document.getElementById('game-end').classList.add('hidden');
    document.getElementById('game-play').classList.remove('hidden');
    currentStepIndex = 0;
    renderStep();
  }

  function renderStep() {
    if (currentStepIndex >= allSteps.length) {
      endGame();
      return;
    }

    const currentStep = allSteps[currentStepIndex];
    document.getElementById('stage-title').innerText = currentStep.phase;
    document.getElementById('feedback').innerText = '';
    document.getElementById('progress').innerText = `مرحله ${currentStepIndex + 1} از ${allSteps.length}`;

    const optionsContainer = document.getElementById('options');
    optionsContainer.innerHTML = '';

    // Create options: 1 correct, 2 incorrect from the same phase or all
    let wrongOptions = allSteps.filter((s, i) => i !== currentStepIndex);
    wrongOptions = shuffleArray(wrongOptions).slice(0, 2);

    let currentOptions = [currentStep, ...wrongOptions];
    currentOptions = shuffleArray(currentOptions);

    currentOptions.forEach(option => {
      const btn = document.createElement('button');
      btn.className = 'option-btn';
      btn.innerText = option.text;
      btn.onclick = () => handleOptionClick(btn, option.id === currentStep.id);
      optionsContainer.appendChild(btn);
    });
  }

  function handleOptionClick(button, isCorrect) {
    const buttons = document.querySelectorAll('.option-btn');
    buttons.forEach(btn => {
      btn.classList.add('disabled');
      btn.onclick = null;
    });

    if (isCorrect) {
      button.classList.add('correct');
      document.getElementById('feedback').innerText = 'آفرین! کاملاً درسته.';
      document.getElementById('feedback').style.color = '#27ae60';
      setTimeout(() => {
        currentStepIndex++;
        renderStep();
      }, 1500);
    } else {
      button.classList.add('incorrect');
      document.getElementById('feedback').innerText = 'اشتباه بود. دوباره سعی کن!';
      document.getElementById('feedback').style.color = '#c0392b';
      setTimeout(() => {
        renderStep(); // Re-render same step
      }, 1500);
    }
  }

  function endGame() {
    document.getElementById('game-play').classList.add('hidden');
    document.getElementById('game-end').classList.remove('hidden');
  }

  function resetGame() {
    startGame();
  }
</script>
